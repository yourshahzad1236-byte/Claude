## DEFINITIONS.TEHSIL_AREA_2
```sql
CREATE OR REPLACE FORCE VIEW DEFINITIONS.TEHSIL_AREA_2 AS
SELECT TEHSIL.TEHSIL_ID,
       UPPER(TEHSIL.NAME) TEHSIL_NAME,
       COUNTRY.COUNTRY_ID,
       COUNTRY.NAME COUNTRY_NAME,
       STATE.STATE_ID,
       STATE.NAME STATE_NAME,
       DISTRICT.DISTRICT_ID,
       DISTRICT.NAME DISTRICT_NAME
  FROM DEFINITIONS.DISTRICT,
       DEFINITIONS.STATE,
       DEFINITIONS.TEHSIL,
       DEFINITIONS.COUNTRY
 WHERE STATE.COUNTRY_ID = DISTRICT.COUNTRY_ID
       AND STATE.STATE_ID = DISTRICT.STATE_ID
       AND DISTRICT.COUNTRY_ID = TEHSIL.COUNTRY_ID
       AND DISTRICT.STATE_ID = TEHSIL.STATE_ID
       AND DISTRICT.DISTRICT_ID = TEHSIL.DISTRICT_ID
       AND COUNTRY.COUNTRY_ID = STATE.COUNTRY_ID;
```

## DEFINITIONS.VU_AD_COMMITTEE_MEMBERS
```sql
CREATE OR REPLACE FORCE VIEW DEFINITIONS.VU_AD_COMMITTEE_MEMBERS AS
SELECT DISTINCT "COMMITTEE_ID", "EMP_CODE", "REMARKS"
  FROM (SELECT M.COMMITTEE_ID, M.EMP_CODE, M.REMARKS
          FROM DEFINITIONS.AD_COMMITTEE_MEMBERS M
         WHERE M.ACTIVE = 'Y'
        UNION ALL
        SELECT MD.COMMITTEE_ID, C.MRNO, MD.REMARKS
          FROM DEFINITIONS.AD_COMMITTEE_DEPARTMENT_WISE MD,
               HRD.CURRENT_EMPLOYEES                    C
         WHERE MD.DEPARTMENT_ID = C.DEPARTMENT_ID
           AND MD.ACTIVE = 'Y'
        UNION ALL
        SELECT MDS.COMMITTEE_ID, V.MRNO, MDS.REMARKS
          FROM DEFINITIONS.AD_COMMITTEE_DESIGNATION_WISE MDS,
               HRD.VU_INFORMATION                        V
         WHERE MDS.DESIGNATION_ID = V.DESIGNATION_ID
           AND V.ACTIVE = 'Y'
           AND V.JOINING_DATE IS NOT NULL
           AND V.LEAVING_DATE IS NULL
           AND MDS.ACTIVE = 'Y'
        UNION ALL
        SELECT MDSC.COMMITTEE_ID, V.MRNO, MDSC.REMARKS
          FROM DEFINITIONS.AD_COMMITTEE_DESIG_CATEG_WISE MDSC,
               HRD.VU_INFORMATION                        V,
               DEFINITIONS.CATEGORY_WISE_DESIGNATION     C
         WHERE MDSC.DESIGNATION_CATEGOTY_ID = C.DESIGNATION_CATEGORY_ID
           AND C.DESIGNATION_ID = V.DESIGNATION_ID
           AND V.ACTIVE = 'Y'
           AND V.JOINING_DATE IS NOT NULL
           AND V.LEAVING_DATE IS NULL
           AND MDSC.ACTIVE = 'Y'
           UNION ALL
        SELECT DP.COMMITTEE_ID, V.MRNO, DP.REMARKS
          FROM DEFINITIONS.Ad_Committee_Dept_Section_Wise DP,
               HRD.VU_INFORMATION                        V
         WHERE DP.DEPARTMENT_ID = V.DEPARTMENT_ID
           AND DP.SECTION_ID = V.SECTION_ID
           AND V.ACTIVE = 'Y'
           AND V.JOINING_DATE IS NOT NULL
           AND V.LEAVING_DATE IS NULL
           AND DP.ACTIVE = 'Y')
 WHERE hrd.pkg_active_directory.f_is_committee_member_exempted(EMP_CODE,
                                                               COMMITTEE_ID) = 'N';
```

## DEFINITIONS.VU_CPT
```sql
CREATE OR REPLACE FORCE VIEW DEFINITIONS.VU_CPT AS
SELECT C.CPT_ID,
       LTRIM(SUBSTR(C.CPT_ID, 4), 0) DISP_CPT_ID,
       C.DESCRIPTION, C.PRICE,

       C.SURGERY_REGION_ID,
       SR.DESCRIPTION SURGERY_REGION,

       C.SPECIALITY_ID,
       CS.DESCRIPTION CLINIC_SPECIALITY,

       C.CPT_CATEGORY_ID CPT_CATEGORY_ID_CURRENT,
       CC.DESCRIPTION PACKAGE_CURRENT,

       C.CPT_CATEGORY_ID_NEW CPT_CATEGORY_ID_NEW,
       CCN.DESCRIPTION PACKAGE_NEW,

       C.INITIALIZED_YN,
       C.INITIALIZED_DATE,
       (SELECT MAX(EFFECTIVE_DATE)
          FROM DEFINITIONS.CPT_PRICE_HISTORY
         WHERE CPT_ID = C.CPT_ID) LAST_EFFECTIVE_DATE,
       C.SHORT_DESC,
       C.LONG_DESC,
       C.ACTIVE,

       CDS.DEPARTMENT_ID,
       D.DESCRIPTION DEPARTMENT,

       CDS.SECTION_ID,
       DS.DESCRIPTION SECTION

  FROM DEFINITIONS.CPT               C,
       DEFINITIONS.CPT_CATEGORY      CCN,
       DEFINITIONS.CPT_CATEGORY      CC,
       DEFINITIONS.SURGERY_REGIONS   SR,
       DEFINITIONS.CLINIC_SPECIALITY CS,
       DEFINITIONS.CPT_DEPARTMENT_SECTION CDS,
       DEFINITIONS.DEPARTMENT_SECTION DS,
       DEFINITIONS.DEPARTMENT D
 WHERE (C.CPT_ID NOT LIKE '%S%'
   and C.CPT_ID NOT LIKE '%O%'
   and C.CPT_ID NOT LIKE '%P%')
   AND C.CPT_CATEGORY_ID_NEW = CCN.CPT_CATEGORY_ID(+)
   AND C.CPT_CATEGORY_ID = CC.CPT_CATEGORY_ID(+)
   AND C.SURGERY_REGION_ID = SR.SURGERY_REGION_ID(+)
   AND C.SPECIALITY_ID = CS.CLINIC_SPECIALITY_ID(+)
   AND C.CPT_ID = CDS.CPT_ID(+)
   AND CDS.DEPARTMENT_ID = DS.DEPARTMENT_ID(+)
   AND CDS.SECTION_ID = DS.SECTION_ID(+)
   AND DS.DEPARTMENT_ID = D.DEPARTMENT_ID(+)
   AND C.CPT_ID IN
   (SELECT CPT_ID
   FROM DEFINITIONS.CPT_DEPARTMENT_SECTION
   WHERE DEPARTMENT_ID='0013500');
```
